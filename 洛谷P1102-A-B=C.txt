#include<iostream>
#include<algorithm>
#include<cstring>

using namespace std;
const int N = 200010; 
int n, C;
int q[N];
int bSearch1(int q[], int len, int x) {
	int l = 0, r = n + 1; 
	while ( 1 + l < r) {
		int mid = (l + r) / 2;
		if (q[mid] < x) {
			l = mid;
		}
		else r = mid;
	}
	if (q[r] == x)return r;
	else return -1;
}
int bSearch2(int q[], int len, int x) {
	int l = 0, r = n + 1;
	while (1 + l < r) {
		int mid = (l + r) / 2;
		if (q[mid] > x) { 
			r = mid;
		}
		else l = mid;
	}
	if (q[l] == x)return l;
	else return -1;
}
int main() {
	memset(q, 0, sizeof q);
	scanf("%d %d", &n, &C);
	for (int i = 1; i <= n; i++) {
		scanf("%lld", &q[i]);
	}
	sort(q + 1, q + 1 + n);
	long long cnt = 0;
	for (int B = 1; B <= n; B++) {
		int A = q[B] + C;
		int res1 = bSearch1(q, n, A);
		if(res1==-1){
			continue;
			}
		else{
			int res2 = bSearch2(q, n, A);
			cnt += res2 - res1 + 1;
		}
	}
	printf("%lld\n", cnt);
	return 0;
}
