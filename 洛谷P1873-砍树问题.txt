#include<iostream>
#include<algorithm>
#include<cstring>
using namespace std;
const int N = 1000010;
int n, m;
int q[N];  
bool check(int x) {
    int sum = 0;
	for (int i = 1; i <= n; i++) {
		sum += max(0, q[i] - x);
       if (sum >= m)return true;
     }
	 return false;
}
int main() {
	scanf("%d%d", &n, &m);
	int highest = 0;
	for (int i = 1; i <= n; i++) {
		scanf("%d", &q[i]);
		highest = max(q[i], highest);
	} 
	int res = 0;
	int sum = 0;
	int l = 0, r = highest+1;
	
	while (l + 1 < r) {
		int mid = (l + r) / 2;
		if (check(mid)) {
			l = mid;
		}
		else r = mid;
	}
	if (check(r))printf("%d\n", r);
	else printf("%d\n", l);
	return 0;
}
